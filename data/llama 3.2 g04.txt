@startuml
skinparam classAttributeIconSize 0

package UserManagement {
  class User {
    +username: String
    +password: String
    +email: String
    -profile: Profile
  }

  class Profile {
    +user_id: Integer
    +name: String
    +address: Address
    +schedule: Schedule
  }

  class Address {
    +address: String
    +latitude: Float
    +longitude: Float
  }

  class Schedule {
    +days_of_week: List<String>
    +hours_of_day: List<String>
  }
}

package DataPackageManagement {
  interface DataProvider {
    +getData(String request): Object
  }

  enum ResponseType {
    public,
    private,
  }

  class FacilityData {
    +name: String
    +address: Address
    +schedule: Schedule
    -hours: Integer
    -acceptable_waste_types: List<String>
  }
}

package Registry {
  class RecyclingFacility {
    +name: String
    +email: String
    -dataProvider: DataProvider
  }

  class Admin {
    +username: String
    +password: String
    -facilities: List<RecyclingFacility>
  }
}

package MapDisplay {
  interface MapService {
    +display(String location): Image
  }

  enum LocationType {
    recycling_center,
    public_bin,
    special_drop_off
  }

  class MapData {
    +name: String
    +latitude: Float
    +longitude: Float
    -type: LocationType
  }
}

package Administration {
  class FacilityAdmin {
    +username: String
    +password: String
    -recycling_facility: RecyclingFacility
  }

  class AdminDashboard {
    +facilities: List<RecyclingFacility>
    +user_stats: Map<String, Integer>
  }
}
@enduml
