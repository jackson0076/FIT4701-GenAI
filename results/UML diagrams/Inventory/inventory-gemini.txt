package "asset" {
    class Asset {
        -String id
        +String name
        +String description
        +String serialNumber
        +String status
        +editAsset(fields: Map<String, String>)
    }

    class Inventory {
        -Asset[] assets
        +addAsset(asset: Asset)
        +updateAsset(asset: Asset)
    }
}

package "request" {
    class Request {
        -String requestId
        -User requester
        -String type
        -RequestStatus status
        -String notificationEmail
        +submit()
        +isPending()
        +notifyUserByEmail()
    }

    enum RequestStatus {
        PENDING
        APPROVED
        DENIED
        EXECUTED
    }

    class TransferRequest extends Request {
        +String fromDepartment
        +String toDepartment
        +approve(approver: User)
    }
}
