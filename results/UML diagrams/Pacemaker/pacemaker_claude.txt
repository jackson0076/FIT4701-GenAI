package "device" {
    class PacemakerDevice {
        - modelNumber: String
        - serialNumber: String
        - implantDate: Date
        + paceSingleChamber(): void
        + paceDualChamber(): void
        + setMode(mode: PacingMode): void
        + getBatteryStatus(): BatteryStatus
        + storeMeasuredData(data: MeasuredData): void
    }

    enum PacingMode {
        OFF, DDDR, VDDR, DDIR, DOOR, VOOR, AOOR, VVIR, AAIR, DDD, VDD, DDI, DOO, VOO, AOO, VVI, AAI, VVT, AAT, OVO, OAO, ODO, OOO
    }

    enum BradycardiaState {
        PERMANENT, TEMPORARY, PACE_NOW, MAGNET, POWER_ON_RESET
    }

    enum BatteryStatus {
        BOL, ERN, ERT, ERP
    }

    class PulseGenerator {
        - atrialAmplitude: Double
        - ventricularAmplitude: Double
        - atrialPulseWidth: Double
        - ventricularPulseWidth: Double
        - lowerRateLimit: Integer
        - upperRateLimit: Integer
        + generatePulse(chamber: Chamber, amplitude: Double, width: Double): void
        + detectMagnet(): Boolean
        + enterPowerOnReset(): void
    }

    enum Chamber {
        ATRIAL, VENTRICULAR
    }

    class LeadSystem {
        - impedance: Double
        - implantDate: Date
        - polarity: String
        + measureImpedance(): Double
        + sense(): Double
        + pace(amplitude: Double, width: Double): void
    }

    class AtrialLead {
        - pacingThreshold: Double
        - pWaveAmplitude: Double
        + measurePWave(): Double
    }

    class VentricularLead {
        - pacingThreshold: Double
        - rWaveAmplitude: Double
        + measureRWave(): Double
    }

    class SensorSystem {
        - activityThreshold: Double
        - responseFactor: Integer
        - maxSensorRate: Integer
        - recoveryTime: Integer
        + detectActivity(): Double
        + calculateRate(): Integer
    }

    class Accelerometer {
        + measureActivity(): Double
        + determineRateIncrease(): Double
        + determineRateDecrease(): Double
    }

    PacemakerDevice --> "1" PulseGenerator
    PacemakerDevice --> "1" SensorSystem
    PacemakerDevice --> "1..2" LeadSystem
    SensorSystem --> "1" Accelerometer
    LeadSystem <|-- AtrialLead
    LeadSystem <|-- VentricularLead
    PacemakerDevice ..> PacingMode
    PacemakerDevice ..> BradycardiaState
    PacemakerDevice ..> BatteryStatus
}
