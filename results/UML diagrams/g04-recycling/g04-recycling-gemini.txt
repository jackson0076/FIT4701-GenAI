package "User Management" {
  class User {
    +userId: String
    +email: String
    +passwordHash: String
    +profile: UserProfile

    +setSchedule(schedule: WeekSchedule): void
    +linkEmail(email: String): void
    +viewMap(): void
    +contactAdmin(message: String): void
    +requestPasswordReset(): void
  }

  class UserProfile {
    +firstName: String
    +lastName: String
    +zipCode: String
    +notificationsEnabled: Boolean
  }

  enum UserRole {

package "Admin & Company" {
  abstract class Admin {
    +adminId: String
    +viewUserFeedback(): List<Feedback>
    +viewUserErrorLogs(): List<ErrorLog>
    +onboardRecyclingCenter(info: RecyclingCenterInfo): void
    +updateFacilityInfo(facilityId: String, info: RecyclingCenterInfo): void
    +viewUsageStats(): UsageStats
    +communicateWithFacilities(message: String): void
    +monitorSiteStatus(): DashboardData
    +blockUser(ipAddress: String): void
  }

  class SystemAdmin extends Admin {
    +manageUsers(): void
    +getSystemData(dataType: String): EncryptedData
  }

  class RecyclingFacilityRep extends User {
    +updateFacilityInfo(info: RecyclingFacilityInfo): void
    +viewUserStats(): UserStats
    +replyToUserQuestion(question: UserQuestion, answer: String): void
    +adjustHours(newHours: List<OperatingHours>): void
    +conveyIssueToAdmin(issue: String): void
  }
}
