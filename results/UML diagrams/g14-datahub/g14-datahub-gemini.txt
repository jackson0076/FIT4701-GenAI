package "users" {
    abstract class Account {
        + signup(provider: string, info: map)
        + login(provider: string, info: map)
        + logout()
        + getProfile(): Profile
        + updateProfile(info: map)
    }

    class User extends Account {
        - name: string
        - email: string
        - is_active: boolean
        - created_at: Date
    }

    class Visitor extends User {
    }

    class Member extends User {
        + member_since: Date
    }

    class Publisher extends Member {
        + configureClient()
    }

    class Admin extends Member {
        + invite(user: User)
        + setSiteConfig(config: map)
        + viewMetrics(): map
        + managePricingPlans()
    }

    class Owner extends Member {
        + inviteMember(user: User)
        + inviteOwner(user: User)
        + removeMember(user: User)
        + setMemberRole(user: User, role: Role)
        + viewMembers(): map
    }

    enum Role {
        MEMBER
        OWNER
    }

    User "1" *-- "0..*" PublisherGroup : owns
    PublisherGroup "1" *-- "1..*" Member : has
    Member "1" --> "1" Role : has

    class PublisherGroup {
        - name: string
        - description: string
        + getMembers(): list<Member>
        + editProfile()
    }
}

package "registry" {
    class Registry {
        + search(query: string): list<DataPackage>
        + getExamplePackages(): list<DataPackage>
        + listPublishers(): list<PublisherGroup>
        + getPublisherProfile(id: string): PublisherGroup
        + getPackageDetails(id: string): DataPackage
        + getPackageVersion(id: string, version: string): DataPackage
        + viewMetrics(): map
        + trackDownload(package: DataPackage)
    }

    class DataPackage {
        - name: string
        - description: string
        - version: string
        - downloads: integer
        + validate(): boolean
        + unpublish()
        + delete()
        + restore()
        + getViews(): list<View>
        + getVersions(): list<DataPackageVersion>
        + getMetrics(): map
    }

    class DataPackageVersion {
        - version: string
        - timestamp: Date
        - schema_changes: map
        + download(): DataPackage
    }

    interface Publishable {
        + create(): DataPackage
        + publish(data: DataPackage)
        + update(data: DataPackage)
        + undelete()
        + tag(tag: string)
        + overwriteTag(tag: string)
    }

    class Client {
        - config: map
        + publish(data: DataPackage)
        + validate(data: DataPackage): boolean
        + preview(data: DataPackage): View
    }

    class View {
        - content: string
    }

    class CachedData {
        + get_data(): Data
    }

    Registry "1" o-- "0..*" DataPackage : contains
    Registry "1" o-- "0..*" PublisherGroup : contains
    Publisher "1" -- "0..*" DataPackage : publishes
    Publisher "1" -- "1" Client : uses
    DataPackage "1" -- "0..*" DataPackageVersion : has_version
    DataPackage "1" -- "0..*" CachedData : caches
    Client "1" -- "0..*" View : previews
    DataPackageVersion "1" --> "1" DataPackage : is_a_version_of
}
